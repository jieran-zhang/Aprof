find_package(ASC REQUIRED)
project(grouped_matmul_direct LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(grouped_matmul op_host/grouped_matmul.asc)
target_include_directories(grouped_matmul PRIVATE op_host op_kernel "$ENV{ASCEND_HOME_PATH}/include")
