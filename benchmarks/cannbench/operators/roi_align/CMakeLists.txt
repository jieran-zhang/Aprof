cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(roi_align_direct LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(roi_align op_host/roi_align.asc)
target_include_directories(roi_align PRIVATE op_host op_kernel)
target_link_libraries(roi_align PRIVATE tiling_api register platform unified_dlog dl m graph_base ascendcl ascendc_runtime)
target_link_directories(roi_align PRIVATE "$ENV{ASCEND_HOME_PATH}/lib64" "$ENV{ASCEND_HOME_PATH}/aarch64-linux/lib64")
target_compile_options(roi_align PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
