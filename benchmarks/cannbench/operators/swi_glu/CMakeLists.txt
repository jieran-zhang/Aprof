cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(swi_glu_direct LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)

add_executable(swi_glu op_host/swi_glu.asc)
target_include_directories(swi_glu PRIVATE op_host op_kernel)
target_link_libraries(swi_glu PRIVATE tiling_api register platform unified_dlog dl m graph_base ascendcl ascendc_runtime)
target_link_directories(swi_glu PRIVATE "$ENV{ASCEND_HOME_PATH}/lib64" "$ENV{ASCEND_HOME_PATH}/aarch64-linux/lib64")
target_compile_options(swi_glu PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
