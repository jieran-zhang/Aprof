0235:                     // 触发条件：
0236:                     //   (1) kBlockIdx == 0：首个 K 块计算时，A1/B1 Pong 尚未使用，可直接搬入
0237:                     //   (2) 当前 L1 大包最后一个 baseK 被消费完，该缓冲区可被覆盖
0238:                     if (((kBlockIdx == 0) || ((kOffsetInChunkB + 1) == stepKb)) && b1NextKChunkIdx < kLoopCount) {
0239:                         AscendC::LocalTensor<half> b1WriteBuf = (b1CopyInIdx == 0) ? b1LocalPing : b1LocalPong;
0240:                         int32_t b1WriteEvent = (b1CopyInIdx == 0) ? EVENT_ID2 : EVENT_ID3;
0241:                         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(b1WriteEvent);
0242:                         DataCopyInB(b1WriteBuf, b1NextKChunkIdx, nBlockIdx);
0243:                         AscendC::SetFlag<AscendC::HardEvent::MTE2_MTE1>(b1WriteEvent);
0244:                         b1NextKChunkIdx += stepKb;
0245:                         b1CopyInIdx ^= 1;
0246:                     }
0247:                     if (((kBlockIdx == 0) || ((kOffsetInChunkA + 1) == stepKa)) && a1NextKChunkIdx < kLoopCount) {
0248:                         AscendC::LocalTensor<half> a1WriteBuf = (a1CopyInIdx == 0) ? a1LocalPing : a1LocalPong;
0249:                         int32_t a1WriteEvent = (a1CopyInIdx == 0) ? EVENT_ID0 : EVENT_ID1;
0250:                         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(a1WriteEvent);
0251:                         DataCopyInA(a1WriteBuf, a1NextKChunkIdx, mBlockIdx);
0252:                         AscendC::SetFlag<AscendC::HardEvent::MTE2_MTE1>(a1WriteEvent);
0253:                         a1NextKChunkIdx += stepKa;
0254:                         a1CopyInIdx ^= 1;
0255:                     }
0256:                 }
0257:                 // ---- CopyOut: CO1 → GM ----
0258:                 CopyOut(cLocal, mBlockIdx, nBlockIdx);
0259:             }
0260:         }
0261: 
0262:         // ============================================================
0263:         // 4. 等待所有同步完成
0264:         // ============================================================
0265:         AscendC::WaitFlag<AscendC::HardEvent::M_MTE1>(EVENT_ID0);
0266:         AscendC::WaitFlag<AscendC::HardEvent::M_MTE1>(EVENT_ID1);
0267:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID0);
0268:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID1);
0269:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID2);
0270:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID3);
0271:     }
0272: 
0273:     __aicore__ inline void InitComputeParams()
0274:     {
0275:         // ---- 1. 计算当前核在 M/N 方向的迭代索引，确定 GM 起始偏移 ----
0276:         constexpr uint32_t mIter = AscendC::Std::ceil_div(M, singleCoreM);
0277:         uint32_t mIterIdx = AscendC::GetBlockIdx() % mIter;
0278:         uint32_t nIterIdx = AscendC::GetBlockIdx() / mIter;
0279: 
0280:         uint64_t gmOffsetA = mIterIdx * singleCoreM * K;
0281:         uint64_t gmOffsetB = IS_B_TRANSPOSE ? nIterIdx * K * singleCoreN : nIterIdx * singleCoreN;
0282:         uint64_t gmOffsetC = mIterIdx * singleCoreM * N + nIterIdx * singleCoreN;
0283:         aGM = aGMOri[gmOffsetA];
0284:         bGM = bGMOri[gmOffsetB];
0285:         cGM = cGMOri[gmOffsetC];
0286: 
0287:         // ---- 2. 计算当前核实际的 M/N 维度大小（最后一块可能不满 singleCore） ----
0288:         actualSingleCoreM = M - mIterIdx * singleCoreM;
0289:         actualSingleCoreM = actualSingleCoreM < singleCoreM ? actualSingleCoreM : singleCoreM;
0290:         actualSingleCoreN = N - nIterIdx * singleCoreN;
0291:         actualSingleCoreN = actualSingleCoreN < singleCoreN ? actualSingleCoreN : singleCoreN;
0292: 
0293:         // ---- 3. 计算 K/M/N 三个维度的循环次数 ----
0294:         kLoopCount = AscendC::Std::ceil_div(singleCoreK, baseK);
0295:         mLoopCount = AscendC::Std::ceil_div(actualSingleCoreM, baseM);
0296:         nLoopCount = AscendC::Std::ceil_div(actualSingleCoreN, baseN);
0297: 
0298:         // ---- 4. 计算 N 方向分块参数：完整 baseN 块数 + 尾块大小及对齐 ----
0299:         baseNCount = actualSingleCoreN / baseN;
0300:         tailN = actualSingleCoreN % baseN;
0301:         tailNAlign = AscendC::Std::ceil_div(tailN, CUBE_BLOCK) * CUBE_BLOCK;
0302: 
0303:         // ---- 5. 计算 M 方向分块参数：完整 baseM 块数 + 尾块大小及对齐 ----
0304:         baseMCount = actualSingleCoreM / baseM;
0305:         tailM = actualSingleCoreM % baseM;
0306:         tailMAlign = AscendC::Std::ceil_div(tailM, CUBE_BLOCK) * CUBE_BLOCK;
0307:     }
0308: 
0309:     // 根据子块索引设置当前子块的全部计算参数
0310:     // mIterIdx: 当前核在 M 方向的子块索引
0311:     // nIterIdx: 当前核在 N 方向的子块索引
0312:     __aicore__ inline void InitComputeParamsL2Cache(uint32_t mIterIdx, uint32_t nIterIdx)
0313:     {
0314:         // ----  计算当前子块在 GM 中的起始偏移 ----
0315:         // aGM/bGM/cGM 指向当前子块起始位置，后续 DataCopyIn/CopyOut 基于该偏移寻址
0316:         uint64_t gmOffsetA = mIterIdx * singleCoreM * K;
0317:         uint64_t gmOffsetB = IS_B_TRANSPOSE ? nIterIdx * K * singleCoreN : nIterIdx * singleCoreN;
0318:         uint64_t gmOffsetC = mIterIdx * singleCoreM * N + nIterIdx * singleCoreN;
0319:         aGM = aGMOri[gmOffsetA];
0320:         bGM = bGMOri[gmOffsetB];
0321:         cGM = cGMOri[gmOffsetC];
0322: 
0323:         actualSingleCoreM = M - mIterIdx * singleCoreM;
0324:         actualSingleCoreM = actualSingleCoreM < singleCoreM ? actualSingleCoreM : singleCoreM;
0325:         actualSingleCoreN = N - nIterIdx * singleCoreN;
0326:         actualSingleCoreN = actualSingleCoreN < singleCoreN ? actualSingleCoreN : singleCoreN;
0327: 
0328:         kLoopCount = AscendC::Std::ceil_div(singleCoreK, baseK);
0329:         mLoopCount = AscendC::Std::ceil_div(actualSingleCoreM, baseM);
0330:         nLoopCount = AscendC::Std::ceil_div(actualSingleCoreN, baseN);
0331:         baseNCount = actualSingleCoreN / baseN;
0332:         tailN = actualSingleCoreN % baseN;
0333:         tailNAlign = AscendC::Std::ceil_div(tailN, CUBE_BLOCK) * CUBE_BLOCK;
0334: 
0335:         baseMCount = actualSingleCoreM / baseM;
0336:         tailM = actualSingleCoreM % baseM;
0337:         tailMAlign = AscendC::Std::ceil_div(tailM, CUBE_BLOCK) * CUBE_BLOCK;
0338:     }
0339: 
0340:     // GM → A1: 将 A 矩阵的一个大包 (stepKa 个 baseK) 搬入 L1
0341:     // kChunkIdx: K 方向大包索引，用于计算 GM 读取偏移
0342:     // mBlockIdx: M 方向 baseM 块索引，用于计算 GM 读取偏移及判断尾块
0343:     __aicore__ inline void DataCopyInA(AscendC::LocalTensor<half> a1Local, uint32_t kChunkIdx, uint32_t mBlockIdx)
0344:     {
0345:         // 尾块使用 tailM，否则使用 baseM
0346:         uint32_t curM = (mBlockIdx != baseMCount) ? baseM : tailM;
0347:         AscendC::Nd2NzParams nd2nzParams;
0348:         nd2nzParams.ndNum = 1;
0349:         nd2nzParams.nValue = curM;
0350:         nd2nzParams.dValue = baseK * stepKa;
0351:         nd2nzParams.srcNdMatrixStride = 0;
0352:         nd2nzParams.srcDValue = K;
0353:         nd2nzParams.dstNzC0Stride = baseM;
0354:         nd2nzParams.dstNzNStride = 1;
0355:         nd2nzParams.dstNzMatrixStride = 0;
0356:         AscendC::DataCopy(a1Local, aGM[kChunkIdx * baseK + mBlockIdx * K * baseM], nd2nzParams);
0357:     }
0358: 
0359:     // GM → B1: 将 B 矩阵的一个大包 (stepKb 个 baseK) 搬入 L1
0360:     // kChunkIdx: K 方向大包索引，用于计算 GM 读取偏移
0361:     // nBlockIdx: N 方向 baseN 块索引，用于计算 GM 读取偏移及判断尾块
0362:     __aicore__ inline void DataCopyInB(AscendC::LocalTensor<half> b1Local, uint32_t kChunkIdx, uint32_t nBlockIdx)
0363:     {
0364:         // 尾块使用 tailN，否则使用 baseN
0365:         uint32_t curN = (nBlockIdx != baseNCount) ? baseN : tailN;
0366:         AscendC::Nd2NzParams nd2nzParams;
0367:         if constexpr (!IS_B_TRANSPOSE) {
0368:             // B 非转置
0369:             nd2nzParams.ndNum = 1;
0370:             nd2nzParams.nValue = baseK * stepKb;
0371:             nd2nzParams.dValue = curN;
0372:             nd2nzParams.srcNdMatrixStride = 0;
0373:             nd2nzParams.srcDValue = N;
0374:             nd2nzParams.dstNzC0Stride = baseK * stepKb;
0375:             nd2nzParams.dstNzNStride = 1;
0376:             nd2nzParams.dstNzMatrixStride = 0;
0377:             AscendC::DataCopy(b1Local, bGM[kChunkIdx * baseK * N + nBlockIdx * baseN], nd2nzParams);
0378:         } else {
0379:             // B 转置
0380:             nd2nzParams.ndNum = 1;
0381:             nd2nzParams.nValue = curN;
0382:             nd2nzParams.dValue = baseK * stepKb;
0383:             nd2nzParams.srcNdMatrixStride = 0;
0384:             nd2nzParams.srcDValue = K;
0385:             nd2nzParams.dstNzC0Stride = baseN;
0386:             nd2nzParams.dstNzNStride = 1;
0387:             nd2nzParams.dstNzMatrixStride = 0;
0388:             AscendC::DataCopy(b1Local, bGM[kChunkIdx * baseK + nBlockIdx * baseN * K], nd2nzParams);
0389:         }
0390:     }
0391: 
0392:     // A1 → A2: 将 L1 中的一个 baseM * baseK 切片搬入 L0
0393:     // mBlockIdx: M 方向 baseM 块索引，用于判断尾块对齐
0394:     // kOffsetInChunkA: 当前 base块 在大包内的偏移序号 (0 ~ stepKa-1)
0395:     __aicore__ inline void DataLoadA(
0396:         AscendC::LocalTensor<half> a1Local, AscendC::LocalTensor<half> a2Local, uint32_t mBlockIdx,
0397:         uint32_t kOffsetInChunkA)
0398:     {
0399:         uint32_t srcAddr = kOffsetInChunkA * baseK * baseM;
0400:         // 尾块使用对齐后的 tailMAlign，保证 LoadData 地址对齐
0401:         uint32_t curMAlign = (mBlockIdx != baseMCount) ? baseM : tailMAlign;
0402: #if defined(__NPU_ARCH__) && (__NPU_ARCH__ == 2201)
0403:         AscendC::LoadData3DParamsV2<half> loadDataParams;
0404:         loadDataParams.l1H = 1;
0405:         loadDataParams.l1W = baseM;
0406:         loadDataParams.channelSize = baseK;
0407:         loadDataParams.kExtension = baseK;
0408:         loadDataParams.mExtension = curMAlign;
0409:         loadDataParams.mStartPt = 0;
0410:         loadDataParams.kStartPt = 0;
0411:         AscendC::LoadData(a2Local, a1Local[srcAddr], loadDataParams);
0412: #elif defined(__NPU_ARCH__) && (__NPU_ARCH__ == 3510)
0413:         AscendC::LoadData2DParamsV2 loadDataParams;
0414:         loadDataParams.mStartPosition = 0;
0415:         loadDataParams.kStartPosition = 0;
0416:         loadDataParams.mStep = AscendC::Std::ceil_div(curMAlign, CUBE_BLOCK);
0417:         loadDataParams.kStep = AscendC::Std::ceil_div(baseK, CUBE_BLOCK);
0418:         loadDataParams.srcStride = AscendC::Std::ceil_div(baseM, CUBE_BLOCK);
0419:         loadDataParams.dstStride = AscendC::Std::ceil_div(curMAlign, CUBE_BLOCK);
0420:         loadDataParams.sid = 0;
0421:         loadDataParams.ifTranspose = false;
0422:         AscendC::LoadData(a2Local, a1Local[srcAddr], loadDataParams);
0423: #endif
0424:     }
0425: 
0426:     // B1 → B2: 将 L1 中的一个 baseK * baseN 切片搬入 L0
0427:     // nBlockIdx: N 方向 baseN 块索引，用于判断尾块对齐
0428:     // kOffsetInChunkB: 当前 base块 在大包内的偏移序号 (0 ~ stepKb-1)
0429:     __aicore__ inline void DataLoadB(
0430:         AscendC::LocalTensor<half> b1Local, AscendC::LocalTensor<half> b2Local, uint32_t nBlockIdx,
0431:         uint32_t kOffsetInChunkB)
0432:     {
0433:         uint32_t srcAddr = kOffsetInChunkB * baseK * (IS_B_TRANSPOSE ? baseN : CUBE_BLOCK);
0434:         // 尾块使用对齐后的 tailNAlign，保证 LoadData 地址对齐
0435:         uint32_t curNAlign = (nBlockIdx != baseNCount) ? baseN : tailNAlign;
0436: #if defined(__NPU_ARCH__) && (__NPU_ARCH__ == 2201)
0437:         if constexpr (!IS_B_TRANSPOSE) {
0438:             // B 非转置: 使用 LoadData（卷积数据搬运）v2 完成 [K, N] → [N, K] 转置搬运
0439:             AscendC::LoadData3DParamsV2<half> loadDataParams;
0440:             loadDataParams.l1H = 1;
0441:             loadDataParams.l1W = baseK * stepKb;
0442:             loadDataParams.channelSize = baseN;
0443:             loadDataParams.kExtension = curNAlign;
0444:             loadDataParams.mExtension = baseK;
0445:             loadDataParams.mStartPt = kOffsetInChunkB * baseK;
0446:             loadDataParams.kStartPt = 0;
0447:             loadDataParams.strideW = 1;
0448:             loadDataParams.strideH = 1;
0449:             loadDataParams.filterW = 1;
0450:             loadDataParams.filterH = 1;
0451:             loadDataParams.dilationFilterW = 1;
0452:             loadDataParams.dilationFilterH = 1;
0453:             loadDataParams.filterSizeW = false;
0454:             loadDataParams.filterSizeH = false;
0455:             loadDataParams.enTranspose = true;
0456:             loadDataParams.fMatrixCtrl = false;
0457:             AscendC::LoadData(b2Local, b1Local, loadDataParams);
0458:         } else {
0459:             // B 转置: 按 CUBE_BLOCK 粒度逐块搬入 L0，无需转置
0460:             AscendC::LoadData2DParams loadDataParams;
0461:             uint32_t dstOffset = curNAlign * CUBE_BLOCK;
0462:             uint32_t srcOffset = baseN * CUBE_BLOCK;
0463:             loadDataParams.repeatTimes = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0464:             loadDataParams.srcStride = 1;
0465:             loadDataParams.dstGap = 0;
0466:             loadDataParams.ifTranspose = false;
0467:             for (int i = 0; i < AscendC::Std::ceil_div(baseK, CUBE_BLOCK); ++i) {
0468:                 AscendC::LoadData(b2Local[i * dstOffset], b1Local[srcAddr + i * srcOffset], loadDataParams);
0469:             }
0470:         }
0471: #elif defined(__NPU_ARCH__) && (__NPU_ARCH__ == 3510)
0472:         if constexpr (!IS_B_TRANSPOSE) {
0473:             // B 非转置: 使用 LoadData2D V2 完成 [K, N] → [N, K] 转置搬运
0474:             AscendC::LoadData2DParamsV2 loadDataParams;
0475:             loadDataParams.mStartPosition = 0;
0476:             loadDataParams.kStartPosition = 0;
0477:             loadDataParams.mStep = AscendC::Std::ceil_div(baseK, CUBE_BLOCK);
0478:             loadDataParams.kStep = AscendC::Std::ceil_div(curNAlign * sizeof(half), 32);
0479:             loadDataParams.srcStride = AscendC::Std::ceil_div(baseK * stepKb, CUBE_BLOCK);
0480:             loadDataParams.dstStride = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0481:             loadDataParams.ifTranspose = true;
0482:             AscendC::LoadData(b2Local, b1Local[srcAddr], loadDataParams);
0483:         } else {
0484:             // B 转置: 无需转置，直接按块搬运
0485:             AscendC::LoadData2DParamsV2 loadDataParams;
0486:             loadDataParams.mStartPosition = 0;
0487:             loadDataParams.kStartPosition = 0;
0488:             loadDataParams.mStep = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0489:             loadDataParams.kStep = AscendC::Std::ceil_div(baseK * sizeof(half), 32);
0490:             loadDataParams.srcStride = AscendC::Std::ceil_div(baseN, CUBE_BLOCK);
0491:             loadDataParams.dstStride = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0492:             loadDataParams.ifTranspose = false;
0493:             AscendC::LoadData(b2Local, b1Local[srcAddr], loadDataParams);
0494:         }
0495: #endif
0496:     }
0497: 
0498:     // A2 * B2 → CO1: 执行一次 Mmad 矩阵乘累加,A矩阵大小baseM * baseK,B矩阵baseK * baseN
0499:     // kBlockIdx: K 方向 baseK 块索引，用于判断是否首次累加
0500:     // mBlockIdx/nBlockIdx: M/N 方向块索引，用于判断尾块实际尺寸
0501:     __aicore__ inline void Compute(
0502:         AscendC::LocalTensor<float> cLocal, AscendC::LocalTensor<half> a2Local, AscendC::LocalTensor<half> b2Local,
0503:         uint32_t kBlockIdx, uint32_t mBlockIdx, uint32_t nBlockIdx)
0504:     {
0505:         // MTE1_M 正向同步：等待 MTE1 完成 L1 → L0 搬运
0506:         AscendC::SetFlag<AscendC::HardEvent::MTE1_M>(mte1DBFlag);
0507:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_M>(mte1DBFlag);
0508:         // 尾块使用 tailM/tailN，否则使用 baseM/baseN
0509:         uint32_t curM = (mBlockIdx != baseMCount) ? baseM : tailM;
0510:         uint32_t curN = (nBlockIdx != baseNCount) ? baseN : tailN;
0511:         AscendC::MmadParams mmadParams;
0512:         mmadParams.m = curM;
0513:         mmadParams.n = curN;
0514:         mmadParams.k = baseK;
0515:         mmadParams.cmatrixInitVal = (kBlockIdx == 0);
0516:         mmadParams.unitFlag = (kBlockIdx != kLoopCount - 1) ? 2 : 3; // 开启UnitFlag
0517:         AscendC::Mmad(cLocal, a2Local, b2Local, mmadParams);
0518:         // M_MTE1 反向同步：通知下一轮 DataLoad 可以覆盖 L0 缓冲区
0519:         AscendC::SetFlag<AscendC::HardEvent::M_MTE1>(mte1DBFlag);
0520:         mte1DBFlag ^= 1;
0521:     }
0522: 
0523:     // CO1 → GM: 将计算结果从 L0C 写回 Global Memory
0524:     // mBlockIdx/nBlockIdx: M/N 方向块索引，用于计算 GM 写入偏移及判断尾块尺寸
0525:     __aicore__ inline void CopyOut(AscendC::LocalTensor<float> cLocal, uint32_t mBlockIdx, uint32_t nBlockIdx)
0526:     {
0527:         // 尾块使用对齐后的 tailMAlign 作为 srcStride，使用 tailM/tailN 作为实际行列数
0528:         uint32_t curMAlign = (mBlockIdx != baseMCount) ? baseM : tailMAlign;
0529:         uint32_t curM = (mBlockIdx != baseMCount) ? baseM : tailM;
0530:         uint32_t curN = (nBlockIdx != baseNCount) ? baseN : tailN;
0531:         AscendC::FixpipeParamsV220 fixpipeParams;
0532:         fixpipeParams.nSize = curN;
0533:         fixpipeParams.mSize = curM;
0534:         fixpipeParams.srcStride = curMAlign;
0535:         fixpipeParams.dstStride = N;
0536:         fixpipeParams.quantPre = QuantMode_t::F322F16;
0537:         fixpipeParams.ndNum = 1;
0538:         fixpipeParams.srcNdStride = 0;
0539:         fixpipeParams.dstNdStride = 0;
0540:         fixpipeParams.unitFlag = 3;
0541:         AscendC::Fixpipe(cGM[mBlockIdx * baseM * N + nBlockIdx * baseN], cLocal, fixpipeParams);
0542:     }

0544: private:
0545:     AscendC::GlobalTensor<half> aGM;
0546:     AscendC::GlobalTensor<half> bGM;
0547:     AscendC::GlobalTensor<half> cGM;
0548:     AscendC::GlobalTensor<half> aGMOri;
0549:     AscendC::GlobalTensor<half> bGMOri;
0550:     AscendC::GlobalTensor<half> cGMOri;
0551:     uint32_t actualSingleCoreM, actualSingleCoreN;
0552:     uint32_t mLoopCount, nLoopCount, kLoopCount;
0553:     uint32_t baseMCount, baseNCount;
0554:     uint32_t tailM, tailN;
0555:     uint32_t tailMAlign, tailNAlign;
0556:     uint8_t mte1DBFlag = 0;
0557: };

0255:                     }
0256:                 }
0257:                 // ---- CopyOut: CO1 → GM ----
0258:                 CopyOut(cLocal, mBlockIdx, nBlockIdx);
0259:             }
0260:         }
0261: 
0262:         // ============================================================
0263:         // 4. 等待所有同步完成
0264:         // ============================================================
0265:         AscendC::WaitFlag<AscendC::HardEvent::M_MTE1>(EVENT_ID0);
0266:         AscendC::WaitFlag<AscendC::HardEvent::M_MTE1>(EVENT_ID1);
0267:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID0);
0268:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID1);
0269:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID2);
0270:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_MTE2>(EVENT_ID3);
0271:     }

0273:     __aicore__ inline void InitComputeParams()
0274:     {
0275:         // ---- 1. 计算当前核在 M/N 方向的迭代索引，确定 GM 起始偏移 ----
0276:         constexpr uint32_t mIter = AscendC::Std::ceil_div(M, singleCoreM);
0277:         uint32_t mIterIdx = AscendC::GetBlockIdx() % mIter;
0278:         uint32_t nIterIdx = AscendC::GetBlockIdx() / mIter;
0279: 
0280:         uint64_t gmOffsetA = mIterIdx * singleCoreM * K;
0281:         uint64_t gmOffsetB = IS_B_TRANSPOSE ? nIterIdx * K * singleCoreN : nIterIdx * singleCoreN;
0282:         uint64_t gmOffsetC = mIterIdx * singleCoreM * N + nIterIdx * singleCoreN;
0283:         aGM = aGMOri[gmOffsetA];
0284:         bGM = bGMOri[gmOffsetB];
0285:         cGM = cGMOri[gmOffsetC];
0286: 
0287:         // ---- 2. 计算当前核实际的 M/N 维度大小（最后一块可能不满 singleCore） ----
0288:         actualSingleCoreM = M - mIterIdx * singleCoreM;
0289:         actualSingleCoreM = actualSingleCoreM < singleCoreM ? actualSingleCoreM : singleCoreM;
0290:         actualSingleCoreN = N - nIterIdx * singleCoreN;
0291:         actualSingleCoreN = actualSingleCoreN < singleCoreN ? actualSingleCoreN : singleCoreN;
0292: 
0293:         // ---- 3. 计算 K/M/N 三个维度的循环次数 ----
0294:         kLoopCount = AscendC::Std::ceil_div(singleCoreK, baseK);
0295:         mLoopCount = AscendC::Std::ceil_div(actualSingleCoreM, baseM);
0296:         nLoopCount = AscendC::Std::ceil_div(actualSingleCoreN, baseN);
0297: 
0298:         // ---- 4. 计算 N 方向分块参数：完整 baseN 块数 + 尾块大小及对齐 ----
0299:         baseNCount = actualSingleCoreN / baseN;
0300:         tailN = actualSingleCoreN % baseN;
0301:         tailNAlign = AscendC::Std::ceil_div(tailN, CUBE_BLOCK) * CUBE_BLOCK;
0302: 
0303:         // ---- 5. 计算 M 方向分块参数：完整 baseM 块数 + 尾块大小及对齐 ----
0304:         baseMCount = actualSingleCoreM / baseM;
0305:         tailM = actualSingleCoreM % baseM;
0306:         tailMAlign = AscendC::Std::ceil_div(tailM, CUBE_BLOCK) * CUBE_BLOCK;
0307:     }
0308: 
0309:     // 根据子块索引设置当前子块的全部计算参数
0310:     // mIterIdx: 当前核在 M 方向的子块索引
0311:     // nIterIdx: 当前核在 N 方向的子块索引
0312:     __aicore__ inline void InitComputeParamsL2Cache(uint32_t mIterIdx, uint32_t nIterIdx)
0313:     {
0314:         // ----  计算当前子块在 GM 中的起始偏移 ----
0315:         // aGM/bGM/cGM 指向当前子块起始位置，后续 DataCopyIn/CopyOut 基于该偏移寻址
0316:         uint64_t gmOffsetA = mIterIdx * singleCoreM * K;
0317:         uint64_t gmOffsetB = IS_B_TRANSPOSE ? nIterIdx * K * singleCoreN : nIterIdx * singleCoreN;
0318:         uint64_t gmOffsetC = mIterIdx * singleCoreM * N + nIterIdx * singleCoreN;
0319:         aGM = aGMOri[gmOffsetA];
0320:         bGM = bGMOri[gmOffsetB];
0321:         cGM = cGMOri[gmOffsetC];
0322: 
0323:         actualSingleCoreM = M - mIterIdx * singleCoreM;
0324:         actualSingleCoreM = actualSingleCoreM < singleCoreM ? actualSingleCoreM : singleCoreM;
0325:         actualSingleCoreN = N - nIterIdx * singleCoreN;
0326:         actualSingleCoreN = actualSingleCoreN < singleCoreN ? actualSingleCoreN : singleCoreN;
0327: 
0328:         kLoopCount = AscendC::Std::ceil_div(singleCoreK, baseK);
0329:         mLoopCount = AscendC::Std::ceil_div(actualSingleCoreM, baseM);
0330:         nLoopCount = AscendC::Std::ceil_div(actualSingleCoreN, baseN);
0331:         baseNCount = actualSingleCoreN / baseN;
0332:         tailN = actualSingleCoreN % baseN;
0333:         tailNAlign = AscendC::Std::ceil_div(tailN, CUBE_BLOCK) * CUBE_BLOCK;
0334: 
0335:         baseMCount = actualSingleCoreM / baseM;
0336:         tailM = actualSingleCoreM % baseM;
0337:         tailMAlign = AscendC::Std::ceil_div(tailM, CUBE_BLOCK) * CUBE_BLOCK;
0338:     }
0339: 
0340:     // GM → A1: 将 A 矩阵的一个大包 (stepKa 个 baseK) 搬入 L1
0341:     // kChunkIdx: K 方向大包索引，用于计算 GM 读取偏移
0342:     // mBlockIdx: M 方向 baseM 块索引，用于计算 GM 读取偏移及判断尾块
0343:     __aicore__ inline void DataCopyInA(AscendC::LocalTensor<half> a1Local, uint32_t kChunkIdx, uint32_t mBlockIdx)
0344:     {
0345:         // 尾块使用 tailM，否则使用 baseM
0346:         uint32_t curM = (mBlockIdx != baseMCount) ? baseM : tailM;
0347:         AscendC::Nd2NzParams nd2nzParams;
0348:         nd2nzParams.ndNum = 1;
0349:         nd2nzParams.nValue = curM;
0350:         nd2nzParams.dValue = baseK * stepKa;
0351:         nd2nzParams.srcNdMatrixStride = 0;
0352:         nd2nzParams.srcDValue = K;
0353:         nd2nzParams.dstNzC0Stride = baseM;
0354:         nd2nzParams.dstNzNStride = 1;
0355:         nd2nzParams.dstNzMatrixStride = 0;
0356:         AscendC::DataCopy(a1Local, aGM[kChunkIdx * baseK + mBlockIdx * K * baseM], nd2nzParams);
0357:     }
0358: 
0359:     // GM → B1: 将 B 矩阵的一个大包 (stepKb 个 baseK) 搬入 L1
0360:     // kChunkIdx: K 方向大包索引，用于计算 GM 读取偏移
0361:     // nBlockIdx: N 方向 baseN 块索引，用于计算 GM 读取偏移及判断尾块
0362:     __aicore__ inline void DataCopyInB(AscendC::LocalTensor<half> b1Local, uint32_t kChunkIdx, uint32_t nBlockIdx)
0363:     {
0364:         // 尾块使用 tailN，否则使用 baseN
0365:         uint32_t curN = (nBlockIdx != baseNCount) ? baseN : tailN;
0366:         AscendC::Nd2NzParams nd2nzParams;
0367:         if constexpr (!IS_B_TRANSPOSE) {
0368:             // B 非转置
0369:             nd2nzParams.ndNum = 1;
0370:             nd2nzParams.nValue = baseK * stepKb;
0371:             nd2nzParams.dValue = curN;
0372:             nd2nzParams.srcNdMatrixStride = 0;
0373:             nd2nzParams.srcDValue = N;
0374:             nd2nzParams.dstNzC0Stride = baseK * stepKb;
0375:             nd2nzParams.dstNzNStride = 1;
0376:             nd2nzParams.dstNzMatrixStride = 0;
0377:             AscendC::DataCopy(b1Local, bGM[kChunkIdx * baseK * N + nBlockIdx * baseN], nd2nzParams);
0378:         } else {
0379:             // B 转置
0380:             nd2nzParams.ndNum = 1;
0381:             nd2nzParams.nValue = curN;
0382:             nd2nzParams.dValue = baseK * stepKb;
0383:             nd2nzParams.srcNdMatrixStride = 0;
0384:             nd2nzParams.srcDValue = K;
0385:             nd2nzParams.dstNzC0Stride = baseN;
0386:             nd2nzParams.dstNzNStride = 1;
0387:             nd2nzParams.dstNzMatrixStride = 0;
0388:             AscendC::DataCopy(b1Local, bGM[kChunkIdx * baseK + nBlockIdx * baseN * K], nd2nzParams);
0389:         }
0390:     }
0391: 
0392:     // A1 → A2: 将 L1 中的一个 baseM * baseK 切片搬入 L0
0393:     // mBlockIdx: M 方向 baseM 块索引，用于判断尾块对齐
0394:     // kOffsetInChunkA: 当前 base块 在大包内的偏移序号 (0 ~ stepKa-1)
0395:     __aicore__ inline void DataLoadA(
0396:         AscendC::LocalTensor<half> a1Local, AscendC::LocalTensor<half> a2Local, uint32_t mBlockIdx,
0397:         uint32_t kOffsetInChunkA)
0398:     {
0399:         uint32_t srcAddr = kOffsetInChunkA * baseK * baseM;
0400:         // 尾块使用对齐后的 tailMAlign，保证 LoadData 地址对齐
0401:         uint32_t curMAlign = (mBlockIdx != baseMCount) ? baseM : tailMAlign;
0402: #if defined(__NPU_ARCH__) && (__NPU_ARCH__ == 2201)
0403:         AscendC::LoadData3DParamsV2<half> loadDataParams;
0404:         loadDataParams.l1H = 1;
0405:         loadDataParams.l1W = baseM;
0406:         loadDataParams.channelSize = baseK;
0407:         loadDataParams.kExtension = baseK;
0408:         loadDataParams.mExtension = curMAlign;
0409:         loadDataParams.mStartPt = 0;
0410:         loadDataParams.kStartPt = 0;

0412: #elif defined(__NPU_ARCH__) && (__NPU_ARCH__ == 3510)
0413:         AscendC::LoadData2DParamsV2 loadDataParams;
0414:         loadDataParams.mStartPosition = 0;
0415:         loadDataParams.kStartPosition = 0;
0416:         loadDataParams.mStep = AscendC::Std::ceil_div(curMAlign, CUBE_BLOCK);
0417:         loadDataParams.kStep = AscendC::Std::ceil_div(baseK, CUBE_BLOCK);
0418:         loadDataParams.srcStride = AscendC::Std::ceil_div(baseM, CUBE_BLOCK);
0419:         loadDataParams.dstStride = AscendC::Std::ceil_div(curMAlign, CUBE_BLOCK);
0420:         loadDataParams.sid = 0;
0421:         loadDataParams.ifTranspose = false;

0423: #endif
0424:     }

0426:     // B1 → B2: 将 L1 中的一个 baseK * baseN 切片搬入 L0
0427:     // nBlockIdx: N 方向 baseN 块索引，用于判断尾块对齐
0428:     // kOffsetInChunkB: 当前 base块 在大包内的偏移序号 (0 ~ stepKb-1)
0429:     __aicore__ inline void DataLoadB(
0430:         AscendC::LocalTensor<half> b1Local, AscendC::LocalTensor<half> b2Local, uint32_t nBlockIdx,
0431:         uint32_t kOffsetInChunkB)
0432:     {
0433:         uint32_t srcAddr = kOffsetInChunkB * baseK * (IS_B_TRANSPOSE ? baseN : CUBE_BLOCK);
0434:         // 尾块使用对齐后的 tailNAlign，保证 LoadData 地址对齐
0435:         uint32_t curNAlign = (nBlockIdx != baseNCount) ? baseN : tailNAlign;
0436: #if defined(__NPU_ARCH__) && (__NPU_ARCH__ == 2201)
0437:         if constexpr (!IS_B_TRANSPOSE) {
0438:             // B 非转置: 使用 LoadData（卷积数据搬运）v2 完成 [K, N] → [N, K] 转置搬运
0439:             AscendC::LoadData3DParamsV2<half> loadDataParams;
0440:             loadDataParams.l1H = 1;
0441:             loadDataParams.l1W = baseK * stepKb;
0442:             loadDataParams.channelSize = baseN;
0443:             loadDataParams.kExtension = curNAlign;
0444:             loadDataParams.mExtension = baseK;
0445:             loadDataParams.mStartPt = kOffsetInChunkB * baseK;
0446:             loadDataParams.kStartPt = 0;
0447:             loadDataParams.strideW = 1;
0448:             loadDataParams.strideH = 1;
0449:             loadDataParams.filterW = 1;
0450:             loadDataParams.filterH = 1;
0451:             loadDataParams.dilationFilterW = 1;
0452:             loadDataParams.dilationFilterH = 1;
0453:             loadDataParams.filterSizeW = false;
0454:             loadDataParams.filterSizeH = false;
0455:             loadDataParams.enTranspose = true;
0456:             loadDataParams.fMatrixCtrl = false;
0457:             AscendC::LoadData(b2Local, b1Local, loadDataParams);
0458:         } else {
0459:             // B 转置: 按 CUBE_BLOCK 粒度逐块搬入 L0，无需转置
0460:             AscendC::LoadData2DParams loadDataParams;
0461:             uint32_t dstOffset = curNAlign * CUBE_BLOCK;
0462:             uint32_t srcOffset = baseN * CUBE_BLOCK;
0463:             loadDataParams.repeatTimes = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0464:             loadDataParams.srcStride = 1;
0465:             loadDataParams.dstGap = 0;
0466:             loadDataParams.ifTranspose = false;
0467:             for (int i = 0; i < AscendC::Std::ceil_div(baseK, CUBE_BLOCK); ++i) {
0468:                 AscendC::LoadData(b2Local[i * dstOffset], b1Local[srcAddr + i * srcOffset], loadDataParams);
0469:             }
0470:         }
0471: #elif defined(__NPU_ARCH__) && (__NPU_ARCH__ == 3510)
0472:         if constexpr (!IS_B_TRANSPOSE) {
0473:             // B 非转置: 使用 LoadData2D V2 完成 [K, N] → [N, K] 转置搬运
0474:             AscendC::LoadData2DParamsV2 loadDataParams;
0475:             loadDataParams.mStartPosition = 0;
0476:             loadDataParams.kStartPosition = 0;
0477:             loadDataParams.mStep = AscendC::Std::ceil_div(baseK, CUBE_BLOCK);
0478:             loadDataParams.kStep = AscendC::Std::ceil_div(curNAlign * sizeof(half), 32);
0479:             loadDataParams.srcStride = AscendC::Std::ceil_div(baseK * stepKb, CUBE_BLOCK);
0480:             loadDataParams.dstStride = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0481:             loadDataParams.ifTranspose = true;
0482:             AscendC::LoadData(b2Local, b1Local[srcAddr], loadDataParams);
0483:         } else {
0484:             // B 转置: 无需转置，直接按块搬运
0485:             AscendC::LoadData2DParamsV2 loadDataParams;
0486:             loadDataParams.mStartPosition = 0;
0487:             loadDataParams.kStartPosition = 0;
0488:             loadDataParams.mStep = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0489:             loadDataParams.kStep = AscendC::Std::ceil_div(baseK * sizeof(half), 32);
0490:             loadDataParams.srcStride = AscendC::Std::ceil_div(baseN, CUBE_BLOCK);
0491:             loadDataParams.dstStride = AscendC::Std::ceil_div(curNAlign, CUBE_BLOCK);
0492:             loadDataParams.ifTranspose = false;
0493:             AscendC::LoadData(b2Local, b1Local[srcAddr], loadDataParams);
0494:         }
0495: #endif
0496:     }

0498:     // A2 * B2 → CO1: 执行一次 Mmad 矩阵乘累加,A矩阵大小baseM * baseK,B矩阵baseK * baseN
0499:     // kBlockIdx: K 方向 baseK 块索引，用于判断是否首次累加
0500:     // mBlockIdx/nBlockIdx: M/N 方向块索引，用于判断尾块实际尺寸
0501:     __aicore__ inline void Compute(
0502:         AscendC::LocalTensor<float> cLocal, AscendC::LocalTensor<half> a2Local, AscendC::LocalTensor<half> b2Local,
0503:         uint32_t kBlockIdx, uint32_t mBlockIdx, uint32_t nBlockIdx)
0504:     {
0505:         // MTE1_M 正向同步：等待 MTE1 完成 L1 → L0 搬运
0506:         AscendC::SetFlag<AscendC::HardEvent::MTE1_M>(mte1DBFlag);
0507:         AscendC::WaitFlag<AscendC::HardEvent::MTE1_M>(mte1DBFlag);
0508:         // 尾块使用 tailM/tailN，否则使用 baseM/baseN
0509:         uint32_t curM = (mBlockIdx != baseMCount) ? baseM : tailM;
0510:         uint32_t curN = (nBlockIdx != baseNCount) ? baseN : tailN;
0511:         AscendC::MmadParams mmadParams;
0512:         mmadParams.m = curM;
0513:         mmadParams.n = curN;
0514:         mmadParams.k = baseK;
0515:         mmadParams.cmatrixInitVal = (kBlockIdx == 0);
0516:         mmadParams.unitFlag = (kBlockIdx != kLoopCount - 1) ? 2 : 3; // 开启UnitFlag
0517:         AscendC::Mmad(cLocal, a2Local, b2Local, mmadParams);
0518:         // M_MTE1 反向同步：通知下一轮 DataLoad 可以覆盖 L0 缓冲区
0519:         AscendC::SetFlag<AscendC::HardEvent::M_MTE1>(mte1DBFlag);
0520:         mte1DBFlag ^= 1;
0521:     }

0523:     // CO1 → GM: 将计算结果从 L0C 写回 Global Memory
0524:     // mBlockIdx/nBlockIdx: M/N 方向块索引，用于计算 GM 写入偏移及判断尾块尺寸

0526:     {

0528:         uint32_t curMAlign = (mBlockIdx != baseMCount) ? baseM : tailMAlign;
0529:         uint32_t curM = (mBlockIdx != baseMCount) ? baseM : tailM;
0530:         uint32_t curN = (nBlockIdx != baseNCount) ? baseN : tailN;
0531:         AscendC::FixpipeParamsV220 fixpipeParams;
0532:         fixpipeParams.nSize = curN;
0533:         fixpipeParams.mSize = curM;
0534:         fixpipeParams.srcStride = curMAlign;
0535:         fixpipeParams.dstStride = N;
0536:         fixpipeParams.quantPre = QuantMode_t::F322F16;
0537:         fixpipeParams.ndNum = 1;
0538:         fixpipeParams.srcNdStride = 0;
0539:         fixpipeParams.dstNdStride = 0;
0540:         fixpipeParams.unitFlag = 3;
0541:         AscendC::Fixpipe(cGM[mBlockIdx * baseM * N + nBlockIdx * baseN], cLocal, fixpipeParams);
0542:     }
