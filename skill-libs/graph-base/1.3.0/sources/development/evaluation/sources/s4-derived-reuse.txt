58: __aicore__ inline float GridReflect(float value, uint64_t size, float sizeFloat, uint32_t alignCorners) {
59:     if (size <= 1) return 0.0f;
60:     const float low = alignCorners ? 0.0f : -0.5f;
61:     const float high = alignCorners ? sizeFloat - 1.0f : sizeFloat - 0.5f;
62:     const float span = high - low;
63:     float distance = value - low;
64:     if (distance < 0.0f) distance = -distance;
65:     int64_t flips = static_cast<int64_t>(distance / span);
66:     float extra = distance - static_cast<float>(flips) * span;
67:     float reflected = (flips & 1LL) == 0 ? extra + low : high - extra;
68:     return GridClamp(reflected, 0.0f, sizeFloat - 1.0f);
69: }

98: __aicore__ inline float GridSampler3DOne(const __gm__ T *x, const __gm__ T *grid,
99:                                           uint64_t linear,
100:                                           const __gm__ GridSampler3DTilingData *t) {
101:         const uint64_t inputSpatial = t->d * t->h * t->w;
102:         const uint64_t spatial = linear % t->spatialOut;
103:         const uint64_t nc = linear / t->spatialOut;
104:         const uint64_t channel = nc % t->c;
105:         const uint64_t batch = nc / t->c;
106:         const uint64_t gridBase = (batch * t->spatialOut + spatial) * 3;
107:         float ix = GridUnnormalize(GridToFloat<T>(grid[gridBase]), t->wFloat, t->alignCorners);
108:         float iy = GridUnnormalize(GridToFloat<T>(grid[gridBase + 1]), t->hFloat, t->alignCorners);
109:         float iz = GridUnnormalize(GridToFloat<T>(grid[gridBase + 2]), t->dFloat, t->alignCorners);
110:         ix = GridApplyPadding(ix, t->w, t->wFloat, t->padding, t->alignCorners);
111:         iy = GridApplyPadding(iy, t->h, t->hFloat, t->padding, t->alignCorners);
112:         iz = GridApplyPadding(iz, t->d, t->dFloat, t->padding, t->alignCorners);
113:         const uint64_t xBase = (batch * t->c + channel) * inputSpatial;
114:         float result = 0.0f;
115:         if (t->interpolation == GRID_NEAREST) {
116:             const int64_t xn = GridRoundEven(ix), yn = GridRoundEven(iy), zn = GridRoundEven(iz);
117:             result = GridRead<T>(x, xBase, zn, yn, xn, t);
118:         } else {
119:             const int64_t x0 = GridFloor(ix), y0 = GridFloor(iy), z0 = GridFloor(iz);
120:             const int64_t x1 = x0 + 1, y1 = y0 + 1, z1 = z0 + 1;
121:             const float dx = ix - static_cast<float>(x0);
122:             const float dy = iy - static_cast<float>(y0);
123:             const float dz = iz - static_cast<float>(z0);
124:             const float wx0 = 1.0f - dx, wy0 = 1.0f - dy, wz0 = 1.0f - dz;
125:             result += GridRead<T>(x, xBase, z0, y0, x0, t) * wx0 * wy0 * wz0;
126:             result += GridRead<T>(x, xBase, z0, y0, x1, t) * dx  * wy0 * wz0;
127:             result += GridRead<T>(x, xBase, z0, y1, x0, t) * wx0 * dy  * wz0;
128:             result += GridRead<T>(x, xBase, z0, y1, x1, t) * dx  * dy  * wz0;
129:             result += GridRead<T>(x, xBase, z1, y0, x0, t) * wx0 * wy0 * dz;
130:             result += GridRead<T>(x, xBase, z1, y0, x1, t) * dx  * wy0 * dz;
131:             result += GridRead<T>(x, xBase, z1, y1, x0, t) * wx0 * dy  * dz;
132:             result += GridRead<T>(x, xBase, z1, y1, x1, t) * dx  * dy  * dz;
133:         }
134:         return result;
135: }
