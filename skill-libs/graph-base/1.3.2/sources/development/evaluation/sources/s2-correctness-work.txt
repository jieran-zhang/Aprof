125: template <typename T>
126: __aicore__ inline void MaximumScalar(GM_ADDR x1, GM_ADDR x2, GM_ADDR y,
127:                                      const __gm__ MaximumTilingData *t) {
128:     __gm__ T *a = reinterpret_cast<__gm__ T *>(x1);
129:     __gm__ T *b = reinterpret_cast<__gm__ T *>(x2);
130:     __gm__ T *out = reinterpret_cast<__gm__ T *>(y);
131:     const uint64_t begin = static_cast<uint64_t>(AscendC::GetBlockIdx()) * t->blockLength;
132:     uint64_t end = begin + t->blockLength;
133:     if (end > t->totalLength) end = t->totalLength;
134:     for (uint64_t linear = begin; linear < end; ++linear) {
135:         uint64_t rem = linear, ai = 0, bi = 0;
136:         for (int dim = static_cast<int>(t->rank) - 1; dim >= 0; --dim) {
137:             const uint64_t coord = rem % t->outShape[dim];
138:             rem /= t->outShape[dim];
139:             ai += coord * t->x1Stride[dim];
140:             bi += coord * t->x2Stride[dim];
141:         }
142:         const T av = a[ai], bv = b[bi];
143:         if constexpr (std::is_same<T, half>::value || std::is_same<T, bfloat16_t>::value) {
144:             const float af = AscendC::ScalarCast<T, float, AscendC::RoundMode::CAST_NONE>(av);
145:             const float bf = AscendC::ScalarCast<T, float, AscendC::RoundMode::CAST_NONE>(bv);
146:             out[linear] = af > bf ? av : bv;
147:         } else {
148:             out[linear] = av > bv ? av : bv;
149:         }
150:     }
151: }

156:     if constexpr (std::is_same<T, int64_t>::value) {

164:     } else {

166:             KernelMaximumSame<T> op; op.Init(x1, x2, y, t); op.Process();

16: struct MaximumTilingData {
17:     uint64_t totalLength;
18:     uint64_t blockLength;
19:     uint64_t outShape[8];
20:     uint64_t x1Stride[8];

29: inline MaximumTilingData ComputeMaximumTiling(
30:     const std::vector<uint64_t> &x1Shape, const std::vector<uint64_t> &x2Shape,
31:     uint32_t dtype, int64_t cores) {
32:     if (x1Shape.empty() || x2Shape.empty() || x1Shape.size() > 8 || x2Shape.size() > 8)
33:         throw std::invalid_argument("rank must be in [1, 8]");
34:     MaximumTilingData t{};
35:     t.rank = static_cast<uint32_t>(std::max(x1Shape.size(), x2Shape.size()));
36:     t.dtype = dtype;
37:     t.tileElements = 4096;
38:     t.sameShape = x1Shape == x2Shape;
39:     std::vector<uint64_t> a(t.rank, 1), b(t.rank, 1);
40:     std::copy(x1Shape.begin(), x1Shape.end(), a.begin() + t.rank - x1Shape.size());
41:     std::copy(x2Shape.begin(), x2Shape.end(), b.begin() + t.rank - x2Shape.size());
42:     uint64_t aStride = 1, bStride = 1, total = 1;
43:     for (int i = static_cast<int>(t.rank) - 1; i >= 0; --i) {
44:         if (a[i] != b[i] && a[i] != 1 && b[i] != 1)
45:             throw std::invalid_argument("incompatible broadcast shapes");
46:         t.outShape[i] = std::max(a[i], b[i]);
47:         t.x1Stride[i] = a[i] == 1 ? 0 : aStride;
48:         t.x2Stride[i] = b[i] == 1 ? 0 : bStride;
49:         aStride *= a[i]; bStride *= b[i]; total *= t.outShape[i];
50:     }
51:     t.totalLength = total;
52:     const uint64_t useful = std::max<uint64_t>(1, (total + 4095) / 4096);
53:     t.blockNum = static_cast<uint32_t>(std::max<int64_t>(1, std::min<int64_t>(cores, useful)));
54:     t.blockLength = (total + t.blockNum - 1) / t.blockNum;
55:     return t;
56: }
