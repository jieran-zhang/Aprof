cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(strided_slice_direct LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)

add_executable(strided_slice op_host/strided_slice.asc)
target_include_directories(strided_slice PRIVATE op_host op_kernel)
target_link_libraries(strided_slice PRIVATE tiling_api register platform unified_dlog dl m graph_base ascendcl ascendc_runtime)
target_link_directories(strided_slice PRIVATE "$ENV{ASCEND_HOME_PATH}/lib64" "$ENV{ASCEND_HOME_PATH}/aarch64-linux/lib64")
target_compile_options(strided_slice PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
