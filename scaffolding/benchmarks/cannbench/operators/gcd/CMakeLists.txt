cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(gcd_direct LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)

add_executable(gcd op_host/gcd.asc)
target_include_directories(gcd PRIVATE op_host op_kernel)
target_link_libraries(gcd PRIVATE tiling_api register platform unified_dlog dl m graph_base ascendcl ascendc_runtime)
target_link_directories(gcd PRIVATE "$ENV{ASCEND_HOME_PATH}/lib64" "$ENV{ASCEND_HOME_PATH}/aarch64-linux/lib64")
target_compile_options(gcd PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
