cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(mhc_sinkhorn_direct LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(mhc_sinkhorn op_host/mhc_sinkhorn.asc)
target_include_directories(mhc_sinkhorn PRIVATE op_host op_kernel)
target_link_libraries(mhc_sinkhorn PRIVATE tiling_api register platform unified_dlog dl m graph_base ascendcl ascendc_runtime)
target_link_directories(mhc_sinkhorn PRIVATE "$ENV{ASCEND_HOME_PATH}/lib64" "$ENV{ASCEND_HOME_PATH}/aarch64-linux/lib64")
target_compile_options(mhc_sinkhorn PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
